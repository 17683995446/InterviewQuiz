// HomeViewModel - 核心状态管理
@HiltViewModel
class HomeViewModel @Inject constructor(
    private val getQuestionsUseCase: GetQuestionsUseCase,
    private val filterQuestionsUseCase: FilterQuestionsUseCase,
    private val getCategoriesUseCase: GetCategoriesUseCase,
    private val getDifficultiesUseCase: GetDifficultiesUseCase,
    private val getQuestionCountUseCase: GetQuestionCountUseCase
) : BaseViewModel() {
    
    private val _uiState = MutableStateFlow(HomeUiState())
    val uiState: StateFlow<HomeUiState> = _uiState.asStateFlow()
    
    init {
        loadQuestions()
    }
    
    fun updateSearchQuery(query: String) {
        _uiState.value = _uiState.value.copy(searchQuery = query)
        applyFilters()
    }
    
    fun selectCategory(category: String?) {
        _uiState.value = _uiState.value.copy(selectedCategory = category)
        applyFilters()
    }
    
    private fun applyFilters() {
        val filtered = filterQuestionsUseCase(
            category = _uiState.value.selectedCategory,
            difficulty = _uiState.value.selectedDifficulty,
            query = _uiState.value.searchQuery
        )
        _uiState.value = _uiState.value.copy(filteredQuestions = filtered)
    }
}

// FilterQuestionsUseCase - 业务用例
class FilterQuestionsUseCase @Inject constructor(
    private val repository: QuestionRepository
) {
    operator fun invoke(
        category: String? = null,
        difficulty: String? = null,
        query: String? = null
    ): List<Question> {
        var questions = repository.getQuestions()
        
        category?.let {
            questions = questions.filter { it.category == category }
        }
        
        difficulty?.let {
            questions = questions.filter { it.difficulty == difficulty }
        }
        
        query?.takeIf { it.isNotBlank() }?.let {
            val lowerQuery = it.lowercase()
            questions = questions.filter { q ->
                q.title.lowercase().contains(lowerQuery) ||
                q.category.lowercase().contains(lowerQuery) ||
                q.explanation.lowercase().contains(lowerQuery) ||
                q.options.any { o -> o.lowercase().contains(lowerQuery) }
            }
        }
        
        return questions
    }
}
